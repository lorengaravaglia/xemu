[built-in options]
# -march=armv8.6-a: the newest ISA level every Thor core (SD 8 Gen 2 "kalama":
#   Cortex-X3/A715/A710/A510, all ARMv9.0-A) runs WITHOUT SVE.  Qualcomm turns
#   SVE off on this SoC (/proc/cpuinfo lists no sve/sve2), and -march=armv9-a
#   enables SVE2 (even "+nosve2+nosve" still defines __ARM_FEATURE_SVE2), so the
#   auto-vectoriser could emit instructions that SIGILL.  armv8.6-a = v9.0 minus
#   SVE: inline LSE atomics (not -moutline-atomics PLT calls), RCPC (LDAPR for
#   acquire loads instead of LDAR), LSE2, dotprod, FRINT, BF16, I8MM -- every
#   one listed in the Thor's /proc/cpuinfo (OPEN-WORK C7, FINDINGS CH).
#   NOTE: this is a hard floor -- the APK will SIGILL on pre-ARMv8.6 devices
#   (e.g. Snapdragon 888 / Cortex-X1).  The Thor is the floor (OPEN-WORK A7).
# -mtune=cortex-x3: kalama's prime core (x4 is Snapdragon 8 Gen 3). Scheduling
#   hint only; does not affect which instructions are legal.
# -fprofile-use: profile-guided optimisation from pgo/xemu.profdata, collected on
#   the Thor from Halo benchmarks and other games' boots (OPEN-WORK C6, FINDINGS
#   CN): vCPU -3%, NV2A thread -7%.  Functions edited since the profile lose it
#   (silently, the warnings are off) -- regenerate with android/tools/pgo-collect.sh.
#   As with the other flags, editing this file does not reach an existing build
#   dir: use android/tools/core-rebuild.sh (CLAUDE.md).
c_args = ['-fPIC', '-DXBOX=1', '-mbranch-protection=none', '-fno-sanitize=shadow-call-stack', '-fno-emulated-tls', '-march=armv8.6-a', '-mtune=cortex-x3', '-fprofile-use=@DIRNAME@/pgo/xemu.profdata', '-Wno-profile-instr-unprofiled', '-Wno-profile-instr-out-of-date', '-Wno-backend-plugin']
